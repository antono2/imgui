cmake_minimum_required(VERSION 3.20)
project(vimgui_accessibility LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_library(vimgui_accessibility STATIC vimgui_accessibility.cpp)
target_compile_features(vimgui_accessibility PUBLIC cxx_std_17)
if(WIN32)
  target_compile_definitions(vimgui_accessibility PRIVATE VIMGUI_ACCESSIBILITY_EXPORTS)
endif()
set_target_properties(vimgui_accessibility PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(vimgui_accessibility PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(vimgui_accessibility PUBLIC Threads::Threads)
if(ANDROID)
  target_sources(vimgui_accessibility PRIVATE android.cpp)
elseif(APPLE)
  enable_language(OBJCXX)
  target_sources(vimgui_accessibility PRIVATE apple.mm)
  set_target_properties(vimgui_accessibility PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
  target_compile_options(vimgui_accessibility PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_compile_definitions(vimgui_accessibility PRIVATE VIMGUI_IOS)
    target_link_libraries(vimgui_accessibility PUBLIC "-framework UIKit" "-framework Foundation" "-framework CoreGraphics")
  else()
    target_link_libraries(vimgui_accessibility PUBLIC "-framework AppKit" "-framework Foundation" "-framework CoreGraphics")
  endif()
elseif(WIN32)
  target_sources(vimgui_accessibility PRIVATE windows.cpp)
  target_link_libraries(vimgui_accessibility PUBLIC uiautomationcore ole32 oleaut32 comctl32)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ATK REQUIRED IMPORTED_TARGET atk atk-bridge-2.0)
  target_sources(vimgui_accessibility PRIVATE linux.cpp)
  target_link_libraries(vimgui_accessibility PUBLIC PkgConfig::ATK)
endif()
option(VIMGUI_ACCESSIBILITY_TESTS "Build retained-tree regression tests" ON)
if(VIMGUI_ACCESSIBILITY_TESTS)
  enable_testing()
  add_executable(accessibility_tree_test ../../tests/accessibility/tree.cpp)
  target_link_libraries(accessibility_tree_test PRIVATE vimgui_accessibility)
  if(MSVC)
    target_compile_options(accessibility_tree_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(accessibility_tree_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME accessibility_tree COMMAND accessibility_tree_test)
  if(WIN32)
    add_executable(accessibility_windows_text_test ../../tests/accessibility/windows_text.cpp)
    target_link_libraries(accessibility_windows_text_test PRIVATE vimgui_accessibility)
    add_test(NAME accessibility_windows_text COMMAND accessibility_windows_text_test)
    add_test(NAME accessibility_windows_text_no_com COMMAND accessibility_windows_text_test --no-com)
    add_test(NAME accessibility_windows_text_managed_mta COMMAND powershell.exe -Mta -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/accessibility/windows_text.ps1" -Executable $<TARGET_FILE:accessibility_windows_text_test>)
    add_test(NAME accessibility_windows_text_managed_sta COMMAND powershell.exe -Sta -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/accessibility/windows_text.ps1" -Executable $<TARGET_FILE:accessibility_windows_text_test>)
    add_test(NAME accessibility_windows_text_system_control COMMAND powershell.exe -Mta -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/accessibility/windows_text.ps1" -Executable $<TARGET_FILE:accessibility_windows_text_test> -SystemControl)
    set_tests_properties(accessibility_windows_text accessibility_windows_text_no_com accessibility_windows_text_managed_mta accessibility_windows_text_managed_sta PROPERTIES TIMEOUT 30)
  endif()
endif()
