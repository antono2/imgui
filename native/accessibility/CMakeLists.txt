cmake_minimum_required(VERSION 3.20)
project(vimgui_accessibility LANGUAGES C CXX)
find_package(Threads REQUIRED)
find_package(ACCESSKIT CONFIG REQUIRED)
set(VIMGUI_ACCESSKIT_STATIC_LIBRARY "" CACHE FILEPATH "Override with a patched AccessKit static library")
if(VIMGUI_ACCESSKIT_STATIC_LIBRARY)
  set_property(TARGET accesskit-static PROPERTY IMPORTED_LOCATION "${VIMGUI_ACCESSKIT_STATIC_LIBRARY}")
else()
  # Release archives live outside an NDK/Apple sysroot; do not re-root this path.
  find_library(VIMGUI_ACCESSKIT_PREBUILT NAMES accesskit
    PATHS "${ACCESSKIT_LIBRARIES_DIR}/static" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  if(NOT VIMGUI_ACCESSKIT_PREBUILT)
    message(FATAL_ERROR "No AccessKit binary for this target. Build from source and set VIMGUI_ACCESSKIT_STATIC_LIBRARY.")
  endif()
  set_property(TARGET accesskit-static PROPERTY IMPORTED_LOCATION "${VIMGUI_ACCESSKIT_PREBUILT}")
endif()
add_library(vimgui_accessibility STATIC vimgui_accessibility.cpp)
target_compile_features(vimgui_accessibility PUBLIC cxx_std_17)
set_target_properties(vimgui_accessibility PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(vimgui_accessibility PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(vimgui_accessibility PUBLIC accesskit-static Threads::Threads ${CMAKE_DL_LIBS})
option(VIMGUI_ACCESSIBILITY_TESTS "Build retained-tree regression tests" ON)
if(VIMGUI_ACCESSIBILITY_TESTS)
  enable_testing()
  add_executable(accessibility_tree_test ../../tests/accessibility/tree.cpp)
  target_link_libraries(accessibility_tree_test PRIVATE vimgui_accessibility)
  if(MSVC)
    target_compile_options(accessibility_tree_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(accessibility_tree_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME accessibility_tree COMMAND accessibility_tree_test)
endif()
